const log = (m) => { document.getElementById("log").textContent += "\n" + m; console.log(m); };
const RATE = 22050;
function wav(buf) {
  const ch = buf.getChannelData(0), n = ch.length;
  const out = new ArrayBuffer(44 + n * 2), v = new DataView(out);
  const str = (o, s) => { for (let i = 0; i < s.length; i++) v.setUint8(o + i, s.charCodeAt(i)); };
  str(0, "RIFF"); v.setUint32(4, 36 + n * 2, true); str(8, "WAVE"); str(12, "fmt ");
  v.setUint32(16, 16, true); v.setUint16(20, 1, true); v.setUint16(22, 1, true);
  v.setUint32(24, RATE, true); v.setUint32(28, RATE * 2, true); v.setUint16(32, 2, true); v.setUint16(34, 16, true);
  str(36, "data"); v.setUint32(40, n * 2, true);
  for (let i = 0; i < n; i++) { const s = Math.max(-1, Math.min(1, ch[i])); v.setInt16(44 + i * 2, s < 0 ? s * 0x8000 : s * 0x7fff, true); }
  return out;
}
(async () => {
  try {
    const names = await (await fetch("list.json")).json();
    for (const name of names) {
      const ab = await (await fetch(name + ".ogg")).arrayBuffer();
      const probe = new OfflineAudioContext(1, 1, RATE);
      const decoded = await probe.decodeAudioData(ab.slice(0));
      // mixdown + resample to RATE mono
      const len = Math.ceil(decoded.duration * RATE);
      const off = new OfflineAudioContext(1, len, RATE);
      const src = off.createBufferSource(); src.buffer = decoded; src.connect(off.destination); src.start();
      const mono = await off.startRendering();
      const res = await fetch("/save?name=" + encodeURIComponent(name), { method: "POST", body: wav(mono) });
      log(`${name}: ${decoded.duration.toFixed(2)}s -> ${await res.text()}`);
    }
    await fetch("/done", { method: "POST" });
    log("DONE");
  } catch (e) { log("ERROR " + e); await fetch("/done?error=" + encodeURIComponent(String(e)), { method: "POST" }); }
})();
</script>
